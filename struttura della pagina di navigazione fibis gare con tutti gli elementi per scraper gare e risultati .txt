const TIPOLOGIA_MAP = {
    'internazionale': { id: '2', btn: '#btn_internazionale', hasComitato: false },
    'fibis challenge': { id: '2', btn: '#btn_internazionale', hasComitato: false },
    'istituzionale': { id: '4', btn: '#btn_istituzionale', hasComitato: true },
    'libera': { id: '5', btn: '#btn_libera', hasComitato: true },
    'riservata': { id: '6', btn: '#btn_riservata', hasComitato: true },
};

// ============================================================
// NAVIGAZIONE PER TIPOLOGIA
// ============================================================

async function navigaPerTipologia(page, tipologia, comitatoId = null) {
    const config = TIPOLOGIA_MAP[tipologia.toLowerCase()];
    if (!config) {
        throw new Error(`Tipologia non supportata: ${tipologia}`);
    }
    
    // Vai alla pagina
    await page.goto('https://www.fibis.it/stecca/fibis-gare-stecca.html', {
        waitUntil: 'networkidle2'
    });
    
    // Seleziona tipologia
    await page.select('#id_tipologia', config.id);
    
    // Seleziona comitato (se necessario e fornito)
    if (config.hasComitato && comitatoId) {
        await page.select('#id_comitato', comitatoId);
    }
    
    // Clicca sul pulsante Cerca specifico
    await page.click(config.btn);
    
    // Aspetta i risultati
    await page.waitForSelector('.current_match', { timeout: 15000 });
    
    // Estrai le gare
    const gare = await page.evaluate(() => {
        return Array.from(document.querySelectorAll('.current_match')).map(item => ({
            titolo: item.querySelector('.info h5')?.textContent?.trim(),
            giorno: item.querySelector('.date_cont .day')?.textContent?.trim(),
            mese: item.querySelector('.date_cont .month')?.textContent?.trim(),
            eventoId: item.querySelector('.date_cont')?.getAttribute('evento-id'),
            luogo: item.querySelector('.loc')?.textContent?.trim(),
            iscrizioni: item.querySelector('.iscrizioni')?.textContent?.trim(),
            haTabellone: !!item.querySelector('.tabelloni'),
            haBatterie: !!item.querySelector('.gironi'),
            haLocandina: !!item.querySelector('.locandina')
        }));
    });
    
    return gare;
}
